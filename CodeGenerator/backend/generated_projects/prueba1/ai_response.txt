import os
from datetime import datetime
from typing import Tuple

def safe_filename(name: str) -> str:
    keep = [c for c in name if c.isalnum() or c in (' ', '.', '_', '-',)]
    s = ''.join(keep).strip().replace(' ', '_')
    return s or datetime.utcnow().strftime('%Y%m%d%H%M%S')

def hash_bytes(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()

def store_upload(base_dir: str, file_bytes: bytes, original_name: str) -> Tuple[str, str]:
    os.makedirs(base_dir, exist_ok=True)
    digest = hash_bytes(file_bytes)[:16]
    clean = safe_filename(original_name)
    fname = f"{digest}__{clean}"
    path = os.path.join(base_dir, fname)
    with open(path, "wb") as f:
        f.write(file_bytes)
    return fname, path

  "backend/app/processing/parser.py": """from typing import List, Dict, Any
import fitz  # PyMuPDF
import re
import logging

logger = logging.getLogger(__name__)

HEADING_RE = re.compile(r'^(Cap[ií]tulo|Tema|Unidad|Secci[oó]n)\\s+\\d+\\b|^[A-ZÁÉÍÓÚÑ][\\w\\s]{0,60}$')
EXERCISE_RE = re.compile(r'^(Ejercicio|Problema|Actividad)\\s*\\d*\\b', re.IGNORECASE)

class ParsedSection:
    def __init__(self, topic: str, subtopic: str | None, text: str, pages: tuple[int, int], kind: str = "content"):
        self.topic = topic
        self.subtopic = subtopic
        self.text = text
        self.pages = pages
        self.kind = kind  # content | exercise


def parse_pdf(path: str) -> List[ParsedSection]:
    doc = fitz.open(path)
    sections: List[ParsedSection] = []
    current_topic = "General"
    current_subtopic = None
    buffer: list[str] = []
    start_page = 0
    def flush(kind="content"):
        nonlocal buffer, start_page, current_topic, current_subtopic
        if buffer:
            text = "\\n".join(buffer).strip()
            if text:
                sections.append(ParsedSection(current_topic, current_subtopic, text, (start_page, page_no), kind))
        buffer = []

    for page_no in range(len(doc)):
        page = doc.load_page(page_no)
        text = page.get_text("text")
        lines = [l.strip() for l in text.splitlines() if l.strip()]
        for line in lines:
            # Detect exercise blocks
            if EXERCISE_RE.match(line):
                flush(kind="content")
                start_page = page_no
                current_subtopic_local = (current_subtopic or current_topic) + " - Ejercicios"
                sections.append(ParsedSection(current_topic, current_subtopic_local, line, (page_no, page_no), "exercise"))
                continue
            # Detect headings
            if HEADING_RE.match(line) and len(line.split()) <= 12:
                flush(kind="content")
                start_page = page_no
                # Heuristic: if all uppercase or short, treat as topic/subtopic
                if current_topic == "General":
                    current_topic = line
                    current_subtopic = None
                else:
                    current_subtopic = line
                continue
            # Accumulate text
            if not buffer:
                start_page = page_no
            buffer.append(line)
        # flush small exercise buffers per page end? Keep accumulating
    flush(kind="content")
    logger.info(f"Parsed {len(sections)} sections from PDF {path}")
    return sections


def parse_file(path: str) -> List[ParsedSection]:
    if path.lower().endswith(".pdf"):
        return parse_pdf(path)
    # Future: add EPUB, DOCX parsing
    raise ValueError(f"Formato no soportado por ahora: {path}")

  "backend/app/processing/chunker.py": """from typing import List, Dict, Any, Tuple
import tiktoken
from .parser import ParsedSection
from ..config import settings
import logging

logger = logging.getLogger(__name__)

def estimate_difficulty(text: str) -> str:
    # Simple heuristic: more formulas and longer sentences => harder
    score = 0
    score += text.count("=") + text.count("\\int") + text.count("\\sum")
    score += text.count("demuest") + text.count("pruebe")
    words = len(text.split())
    if score >= 5 or words > 400:
        return "hard"
    if score >= 2 or words > 200:
        return "medium"
    return "easy"

def tokenize_len(text: str, model: str = "cl100k_base") -> int:
    try:
        enc = tiktoken.get_encoding(model)
    except Exception:
        enc = tiktoken.get_encoding("cl100k_base")
    return len(enc.encode(text))

def chunk_section(section: ParsedSection, max_tokens: int, overlap: int) -> List[Dict[str, Any]]:
    text = section.text
    # naive sentence split
    sentences = [s.strip() for s in text.replace("\\n", " ").split(". ") if s.strip()]
    chunks = []
    curr = ""
    for s in sentences:
        candidate = (curr + " " + s).strip()
        if tokenize_len(candidate) > max_tokens:
            if curr:
                chunks.append(curr.strip())
            curr = s
        else:
            curr = candidate
    if curr:
        chunks.append(curr.strip())
    # apply overlap by merging previous tail
    final_chunks: List[str] = []
    prev = ""
    for i, c in enumerate(chunks):
        if prev and overlap > 0:
            tail_words = prev.split()[-min(len(prev.split()), 50):]
            c = " ".join(tail_words) + " " + c
        final_chunks.append(c)
        prev = c
    results = []
    for c in final_chunks:
        results.append({
            "topic": section.topic,
            "subtopic": section.subtopic,
            "difficulty": estimate_difficulty(c) if section.kind != "exercise" else "medium",
            "text": c,
            "pages": section.pages,
            "kind": section.kind
        })
    return results

def build_chunks(sections: List[ParsedSection]) -> List[Dict[str, Any]]:
    mx = settings.MAX_CHUNK_TOKENS
    ov = settings.CHUNK_OVERLAP_TOKENS
    out: List[dict[str, Any]] = []
    for sec in sections:
        out.extend(chunk_section(sec, mx, ov))
    logger.info(f"Built {len(out)} chunks from {len(sections)} sections")
    return out

from langchain_openai import ChatOpenAI
from ..config import settings
from ..retrieval.retriever import retrieve_by_topics
from ..models.schemas import ExamResponse, Question
import logging

FALLBACK_MC_OPTIONS = ["A", "B", "C", "D"]

def use_openai() -> bool:
    return bool(settings.OPENAI_API_KEY and settings.LLM_MODEL)

def build_prompt(context: str, topic: str, difficulty: str, num_q: int) -> str:
    template = """Eres un generador de exámenes para estudiantes.
Crea {num_q} preguntas de opción múltiple de dificultad {difficulty} sobre el tema: "{topic}".
Usa únicamente el siguiente contexto del libro, preserva notación y precisión.
Proporciona JSON con formato:
[
  {{"id": "uuid", "topic": "string", "prompt": "string", "options": ["A) ...","B) ...","C) ...","D) ..."], "answer": "A|B|C|D", "explanation": "string", "difficulty":"{difficulty}"}},
  ...
]
Contexto:
{context}
"""
    return PromptTemplate.from_template(template).format(topic=topic, difficulty=difficulty, context=context[:8000], num_q=num_q)

def generate_exam(book_id: int, topics: List[str], difficulty: str, num_questions: int) -> ExamResponse:
    per_topic = max(1, num_questions // max(1, len(topics)))
    questions: List[Question] = []
    for t in topics:
        docs = retrieve_by_topics(book_id, [t], k=6)
        context = "\\n\\n".join([d.page_content for d in docs])
        if use_openai():
            try:
                llm = ChatOpenAI(model=settings.LLM_MODEL, api_key=settings.OPENAI_API_KEY, temperature=0.3)
                prompt = build_prompt(context, t, difficulty, per_topic)
                resp = llm.invoke(prompt)
                import json
                data = json.loads(resp.content)
                for q in data[:per_topic]:
                    questions.append(Question(
                        id=q.get("id") or str(uuid.uuid4()),
                        topic=q.get("topic") or t,
                        prompt=q.get("prompt") or "",
                        options=q.get("options"),
                        answer=q.get("answer"),
                        explanation=q.get("explanation"),
                        difficulty=q.get("difficulty") or difficulty
                    ))
                continue
            except Exception as e:
                logger.warning(f"Fallo LLM, usando fallback: {e}")
        # Fallback: simple generation from context
        lines = [ln.strip() for ln in context.splitlines() if len(ln.strip()) > 20]
        for i in range(min(per_topic, len(lines))):
            base = lines[i][:180].rstrip(".")
            prompt = f"{base} ¿Cuál es la opción correcta?"
            options = [f"{opt}) {base}." for opt in ["A", "B", "C", "D"]]
            answer = "A"
            questions.append(Question(
                id=str(uuid.uuid4()),
                topic=t,
                prompt=prompt,
                options=options,
                answer=answer,
                explanation="Basado en el contexto proporcionado.",
                difficulty=difficulty
            ))
    # If still not enough, pad with generic
    while len(questions) < num_questions:
        questions.append(Question(
            id=str(uuid.uuid4()),
            topic=topics[0] if topics else "General",
            prompt="Pregunta de ejemplo: seleccione la opción correcta.",
            options=[f"{o}) Opción {o}" for o in ["A", "B", "C", "D"]],
            answer="A",
            explanation="Ejemplo.",
            difficulty=difficulty
        ))
    return ExamResponse(title="Examen Personalizado", questions=questions[:num_questions])

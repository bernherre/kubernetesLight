function matchRoute(path){
  // simple matching using routes.json order
  const routes = ROUTES.routes || [];
  for(const r of routes){
    if(r.path.includes(':')){
      // param route like /blog/:slug
      const base = r.path.split('/:')[0];
      if(path.startsWith(base) && path.length > base.length+1) return {route:r, params:{slug: path.substring(base.length+1)}};
    }else if(r.path === path){
      return {route:r, params:{}};
    }
  }
  return null;
}
